def is_perfect_square(n: int) -> bool:
    if n < 0:
        return False  # số âm không phải số chính phương
    sqrt_n = int(math.isqrt(n))  # isqrt lấy căn bậc 2 nguyên chính xác
    return sqrt_n * sqrt_n == n
